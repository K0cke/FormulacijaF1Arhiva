const $ = id => document.getElementById(id);

const DESC = {
  Kvalifikacije: "Cela kvalifikaciona sesija, od prvog do poslednjeg kruga.",
  Sprint: "Sprint trka od starta do cilja, sa svim duelima.",
  Trka: "Cela trka od formacionog kruga do kockaste zastave."
};

const COL = {
  Kvalifikacije: ["#2a0509", "#6a0610"],
  Sprint: ["#7a0610", "#c40a14"],
  Trka: ["#9b000c", "#e10600"]
};

const st = {
  season: "",
  gp: "",
  commentator: ""
};

function getSeasons() {
    return [...new Set(
        VIDEOS.map(v => v.season)
    )].sort((a,b) => b-a);
}

function populateSeasons() {

  const seasons = [
    ...new Set(
      VIDEOS.map(v => v.season)
    )
  ].sort((a,b) => b - a);

  $("season").innerHTML = seasons
    .map(s => `<option>${s}</option>`)
    .join("");

  st.season = seasons[0];
}

function populateGP() {

    const gps = VIDEOS
        .filter(v => v.season === st.season)
        .map(v => v.gp);

    const unique = [...new Set(gps)];

    $("gp").innerHTML =
        `<option value="">Sve Velike nagrade</option>` +

        unique
            .map(gp => `
                <option value="${gp}">
                    ${gp}
                </option>
            `)
            .join("");
}

function renderCards() {

    const filtered = VIDEOS.filter(video => {

        return (
            video.season === st.season &&
            (!st.gp || video.gp === st.gp)
        );

    });

    $("grid").innerHTML = filtered.map(video => {

        const colors =
            COL[video.session] ||
            ["#9b000c", "#e10600"];

        return `
            <button
                class="card"
                type="button"
                onclick="window.open('${video.link}','_blank')">

                <div
                    class="thumb"
                    style="
                        --c1:${colors[0]};
                        --c2:${colors[1]};
                    ">

                    <span class="tag">
                        ${video.session}
                    </span>

                    <b>${video.gp}</b>

                </div>

                <div class="info">

                    <h3>
                        VN ${video.gp} ${video.season}
                    </h3>

                    <p>
                        ${DESC[video.session]}
                    </p>

                    <p
                        style="
                            color:#e10600;
                            font-weight:600;
                        ">

                        Komentator:
                        ${video.commentator}

                    </p>

                    <div class="meta">

                        <span>
                            ${video.date}
                        </span>

                        <em>
                            ${video.duration}
                        </em>

                    </div>

                </div>

            </button>
        `;

    }).join("");
}

populateSeasons();
populateGP();
renderCards();

$("season").onchange = e => {

    st.season = e.target.value;

    populateGP();

    renderCards();

};
